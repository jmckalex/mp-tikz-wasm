% tikzlibrarysvg.attributes.code.tex -- ids, classes and other attributes on
% the SVG that PGF's dvisvgm driver writes. Part of mp-tikz-wasm (LGPL-3.0).
%
%   \usetikzlibrary{svg.attributes}
%   \begin{scope}[svg class=fragment] ... \end{scope}
%   \draw[svg id=axis, svg class={thin grey}] ...;
%   \node[svg attributes={data-fragment-index="2"}] {...};
%
% Keys, in any scope, path or node options (and in styles such as every node):
%   svg class=<classes>       added to the element's class list; repeatable
%   svg id=<id>               the element's id (the last one given wins)
%   svg attributes=<text>     name="value" pairs, separated by spaces or commas
%
% How: PGF opens an "id scope" <g> around every scope, path and node, and the
% SVG drivers write \pgfsys@svg@rdf into its start tag -- the hook TikZ's rdf
% library uses for RDFa. These keys collect attributes and hand them to that
% hook, so they land on PGF's own <g>, which PGF also closes: no raw specials
% and nothing to balance. Under a driver that is not SVG (pdfTeX, dvips) the
% hook does not exist and the keys do nothing, so one source builds anywhere.
%
% Not supported: a clip-only path (\clip ...;), which PGF gives no <g> -- its
% attributes would land on the next element instead, so they are dropped.
% With PGF animations on the same element, PGF writes its own id too; use
% svg class there. Executing `rdf engine on` after loading this library
% replaces the hook and turns these keys off.

\ProvidesFileRCS{tikzlibrarysvg.attributes.code.tex}

\let\tikz@svgattr@class\pgfutil@empty
\let\tikz@svgattr@id\pgfutil@empty
\let\tikz@svgattr@raw\pgfutil@empty

\tikzset{
  svg class/.code={%
    \ifx\tikz@svgattr@class\pgfutil@empty
      \def\tikz@svgattr@class{#1}%
    \else
      \expandafter\def\expandafter\tikz@svgattr@class\expandafter{\tikz@svgattr@class\space#1}%
    \fi},
  svg id/.code={\def\tikz@svgattr@id{#1}},
  svg attributes/.code={\tikz@svgattr@addraw{#1}},
}

% svg attributes: name="value" pairs, separated by spaces or -- the pgfkeys habit
% -- commas. Written straight into the start tag, a comma is malformed XML and
% dvisvgm aborts, so the value is detokenized (every character becomes plain text,
% so an active " under babel cannot bite) and scanned: commas outside quotes become
% spaces, quoted values are copied untouched (aria-label="a, b" keeps its comma).
% Unbalanced quotes are a TeX error here rather than a dvisvgm crash later.
\newif\iftikz@svgattr@odd
\def\tikz@svgattr@stop{\tikz@svgattr@stop}% delimiters, never expanded
\def\tikz@svgattr@eos{\tikz@svgattr@eos}
\def\tikz@svgattr@eosm{\tikz@svgattr@eos}
\def\tikz@svgattr@addraw#1{%
  \edef\tikz@svgattr@in{#1}%
  \edef\tikz@svgattr@in{\detokenize\expandafter{\tikz@svgattr@in}}%
  \tikz@svgattr@oddfalse
  \expandafter\tikz@svgattr@countq\tikz@svgattr@in"\tikz@svgattr@stop
  \iftikz@svgattr@odd
    \pgfutil@packageerror{tikz}{svg attributes: unbalanced double quotes in `\tikz@svgattr@in'}%
      {Write the attributes as name="value" pairs, separated by spaces or commas.}%
  \else
    \let\tikz@svgattr@norm\pgfutil@empty
    \expandafter\tikz@svgattr@out\tikz@svgattr@in"\tikz@svgattr@eos"%
    \edef\tikz@svgattr@raw{\tikz@svgattr@raw\space\tikz@svgattr@norm}%
  \fi}
% count the double quotes (the one appended by the caller is not counted)
\def\tikz@svgattr@countq#1"#2\tikz@svgattr@stop{%
  \def\tikz@svgattr@rest{#2}%
  \ifx\tikz@svgattr@rest\pgfutil@empty\else
    \iftikz@svgattr@odd\tikz@svgattr@oddfalse\else\tikz@svgattr@oddtrue\fi
    \tikz@svgattr@countq#2\tikz@svgattr@stop
  \fi}
% alternate: text outside quotes (commas become spaces), then a quoted value
\def\tikz@svgattr@out#1"#2"{%
  \tikz@svgattr@commas{#1}%
  \def\tikz@svgattr@rest{#2}%
  \ifx\tikz@svgattr@rest\tikz@svgattr@eosm\else
    \edef\tikz@svgattr@norm{\tikz@svgattr@norm"#2"}%
    \expandafter\tikz@svgattr@out
  \fi}
\def\tikz@svgattr@commas#1{\tikz@svgattr@c#1,\tikz@svgattr@stop}
\def\tikz@svgattr@c#1,#2\tikz@svgattr@stop{%
  \edef\tikz@svgattr@norm{\tikz@svgattr@norm#1}%
  \def\tikz@svgattr@rest{#2}%
  \ifx\tikz@svgattr@rest\pgfutil@empty\else
    \edef\tikz@svgattr@norm{\tikz@svgattr@norm\space}%
    \tikz@svgattr@c#2\tikz@svgattr@stop
  \fi}

% Hand the collected attributes to the next id scope, then forget them so that
% nothing inside inherits them. \pgfsys@svg@rdf is global (the driver resets it
% as it opens the <g>); what is already in it (RDFa) is kept in front.
\def\tikz@svgattr@flush{%
  \ifx\pgfsys@svg@rdf\pgfutil@undefined\else
    \tikz@svgattr@compose
    \ifx\tikz@svgattr@all\pgfutil@empty\else
      \toks0=\expandafter{\pgfsys@svg@rdf}%
      \toks2=\expandafter{\tikz@svgattr@all}%
      \xdef\pgfsys@svg@rdf{\the\toks0 \the\toks2}%
    \fi
  \fi
  \tikz@svgattr@clear}
\def\tikz@svgattr@compose{%
  \let\tikz@svgattr@all\pgfutil@empty
  \ifx\tikz@svgattr@id\pgfutil@empty\else
    \edef\tikz@svgattr@all{\space id="\tikz@svgattr@id"}\fi
  \ifx\tikz@svgattr@class\pgfutil@empty\else
    \edef\tikz@svgattr@all{\tikz@svgattr@all\space class="\tikz@svgattr@class"}\fi
  \ifx\tikz@svgattr@raw\pgfutil@empty\else
    \edef\tikz@svgattr@all{\tikz@svgattr@all\tikz@svgattr@raw}\fi}
\def\tikz@svgattr@clear{%
  \let\tikz@svgattr@class\pgfutil@empty
  \let\tikz@svgattr@id\pgfutil@empty
  \let\tikz@svgattr@raw\pgfutil@empty}

% The hook TikZ calls before every id scope. Keeping the rdf library's
% behaviour (its option lists are empty unless it is loaded and switched on)
% means loading both works.
\ifx\tikz@rdf@pre@options\pgfutil@undefined \let\tikz@rdf@pre@options\pgfutil@empty \fi
\ifx\tikz@rdf@post@options\pgfutil@undefined \let\tikz@rdf@post@options\pgfutil@empty \fi
\def\tikz@clear@rdf@options{%
  \let\tikz@rdf@pre@options\pgfutil@empty
  \let\tikz@rdf@post@options\pgfutil@empty
  \tikz@svgattr@clear}
\def\tikz@do@rdf@pre@options{%
  \tikz@rdf@pre@options\let\tikz@rdf@pre@options\pgfutil@empty
  % a clip path gets no id scope (tikz.code.tex, \tikz@finish): drop, do not leak
  \iftikz@mode@clip\tikz@svgattr@clear\else\tikz@svgattr@flush\fi}
\def\tikz@do@rdf@post@options{%
  \tikz@rdf@post@options\let\tikz@rdf@post@options\pgfutil@empty}

\endinput
